SELECT temp.activity_date AS day, COUNT(DISTINCT temp.user_id) AS active_users FROM (
    SELECT * FROM Activity 
WHERE activity_date BETWEEN '2019-06-28' AND '2019-07-27'
) AS temp 
GROUP BY temp.activity_date;
